cmake_minimum_required(VERSION 3.16)
project(TestAIConnection)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find Qt6 components (only basic ones)
find_package(Qt6 REQUIRED COMPONENTS Core)

qt6_standard_project_setup()

# Add GPT4All sources
set(GPT4ALL_SOURCES
    gpt4all-chat/src/chat.cpp
    gpt4all-chat/src/chatllm.cpp
    gpt4all-chat/src/chatmodel.cpp
    gpt4all-chat/src/database.cpp
    gpt4all-chat/src/modellist.cpp
    gpt4all-chat/src/mysettings.cpp
    gpt4all-chat/src/logger.cpp
    gpt4all-chat/src/network.cpp
    gpt4all-chat/src/llm.cpp
)

# Create test executable
add_executable(test_ai_connection
    test_ai_connection.cpp
    ${GPT4ALL_SOURCES}
)

# Link libraries
target_link_libraries(test_ai_connection
    Qt6::Core
)

# Include directories
target_include_directories(test_ai_connection PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/gpt4all-chat/src
    ${CMAKE_CURRENT_SOURCE_DIR}/gpt4all-backend
)
