cmake_minimum_required(VERSION 3.16)
project(RealGPT4AllAPIServer)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find Qt6 components
find_package(Qt6 REQUIRED COMPONENTS Core Network HttpServer)

qt6_standard_project_setup()

# Add GPT4All sources needed for the server
set(GPT4ALL_SOURCES
    gpt4all-chat/src/chat.cpp
    gpt4all-chat/src/chatllm.cpp
    gpt4all-chat/src/chatmodel.cpp
    gpt4all-chat/src/database.cpp
    gpt4all-chat/src/modellist.cpp
    gpt4all-chat/src/mysettings.cpp
    gpt4all-chat/src/logger.cpp
    gpt4all-chat/src/network.cpp
    gpt4all-chat/src/llm.cpp
)

# Add executable
add_executable(real_gpt4all_server 
    real_gpt4all_server.cpp
    ${GPT4ALL_SOURCES}
)

# Qt auto-generated files
qt6_add_resources(real_gpt4all_server "config"
    PREFIX "/"
    BASE gpt4all-chat/
    FILES 
        gpt4all-chat/icons/gpt4all.svg
)

# Link libraries
target_link_libraries(real_gpt4all_server 
    Qt6::Core 
    Qt6::Network 
    Qt6::HttpServer
)

# Include directories
target_include_directories(real_gpt4all_server PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/gpt4all-chat/src
    ${CMAKE_CURRENT_SOURCE_DIR}/gpt4all-backend
)

# Compiler definitions
target_compile_definitions(real_gpt4all_server PRIVATE
    QT_DISABLE_DEPRECATED_BEFORE=0x060000
)
